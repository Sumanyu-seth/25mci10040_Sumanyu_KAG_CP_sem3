class Solution:
    def waysToReachTarget(self, target: int, types: list[list[int]]) -> int:
        MOD = 10**9 + 7
        dp = [0] * (target + 1)
        dp[0] = 1
        for count, marks in types:
            new = [0] * (target + 1)
            span = (count + 1) * marks
            for t in range(target + 1):
                v = dp[t]
                if t >= marks:
                    v += new[t - marks]
                if t >= span:
                    v -= dp[t - span]
                new[t] = v % MOD
            dp = new
        return dp[target]
