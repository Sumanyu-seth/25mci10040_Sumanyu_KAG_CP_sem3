import heapq
class Solution:
    def countPaths(self, n: int, roads: list[list[int]]) -> int:
        MOD = 10**9 + 7
        graph = [[] for _ in range(n)]
        for u,v,w in roads:
            graph[u].append((v,w))
            graph[v].append((u,w))
        
        dist = [float('inf')]*n
        ways = [0]*n
        dist[0] = 0
        ways[0] = 1
        pq = [(0,0)]
        while pq:
            d, u = heapq.heappop(pq)
            if d>dist[u]:
                continue

            for v, weigh in graph[u]:
                newdist = dist[u]+weigh
                if newdist < dist[v]:
                    dist[v] = newdist
                    ways[v] = ways[u]
                    heapq.heappush(pq, (newdist, v))
                elif newdist == dist[v]:
                    ways[v] = (ways[v]+ways[u]) % MOD
                    
        return ways[n-1]
