class Solution:
    def removeStones(self, stones: List[List[int]]) -> int:
        ROWS = max(i[0] for i in stones)+1
        COLS = max(i[1] for i in stones)+1
        print(ROWS, COLS)
        par = list(range(ROWS + COLS))
        print(par)

        def find(i):
            print("find", i)
            if par[i] != i:
                par[i] = find(par[i])
            return par[i]

        def merge(i, j):
            i, j = find(i), find(j)

            if i == j:
                return 0

            par[i] = j
            return 1

        for x, y in stones:
            merge(x, ROWS + y)

        components = set()

        for x, y in stones:
            components.add(find(x))
            components.add(find(ROWS + y))

        return len(stones) - len(components)
